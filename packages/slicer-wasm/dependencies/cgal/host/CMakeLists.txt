cmake_minimum_required(VERSION 3.20)
project(orca_gmp_generators C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
foreach(name bases fac fib jacobitab psqr trialdivtab)
  add_executable(gen-${name} "${GMP_SOURCE}/gen-${name}.c")
  if(NOT MSVC)
    target_link_libraries(gen-${name} PRIVATE m)
  endif()
endforeach()
