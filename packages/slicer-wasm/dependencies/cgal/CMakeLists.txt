# Read the library inventories from the pinned upstream Automake inputs.
# Only mpn's configure-selected generic-C aliases need a separate manifest.
function(am_sources output directory variable)
  file(READ "${directory}/Makefile.am" text)
  string(REPLACE "\\\n" " " text "${text}")
  string(REGEX MATCH "(^|\n)${variable}[ \t]*=[^\n]*" line "${text}")
  if(NOT line)
    message(FATAL_ERROR "Missing upstream source inventory ${variable}")
  endif()
  string(REGEX MATCHALL "[A-Za-z0-9_./-]+\\.(c|cc)([ \t]|$)" names "${line}")
  set(result)
  foreach(name IN LISTS names)
    string(STRIP "${name}" name)
    list(APPEND result "${directory}/${name}")
  endforeach()
  set(${output} "${result}" PARENT_SCOPE)
endfunction()

file(MAKE_DIRECTORY "${gmp_generated}/mpn")

set(LIBGMP_DLL 0)
set(CC emcc)
set(CFLAGS "-m${WASM_WIDTH}")
configure_file("${GMP_SOURCE}/gmp-h.in" "${gmp_generated}/gmp.h" @ONLY)

am_sources(gmp_sources "${GMP_SOURCE}" libgmp_la_SOURCES)
list(APPEND gmp_sources "${GMP_SOURCE}/tal-reent.c")
foreach(dir mpf mpz mpq printf scanf rand)
  if(dir STREQUAL "rand")
    set(variable librandom_la_SOURCES)
  else()
    set(variable lib${dir}_la_SOURCES)
  endif()
  am_sources(component "${GMP_SOURCE}/${dir}" ${variable})
  list(APPEND gmp_sources ${component})
endforeach()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/gmp-mpn.json" mpn)
string(JSON count LENGTH "${mpn}")
math(EXPR last "${count} - 1")
foreach(i RANGE 0 ${last})
  string(JSON name MEMBER "${mpn}" ${i})
  string(JSON source GET "${mpn}" "${name}")
  if(source MATCHES "^generated/")
    string(REPLACE "generated/" "${GMP_TABLES}/mpn/" source "${source}")
  else()
    set(source "${GMP_SOURCE}/${source}")
  endif()
  set(wrapper "${gmp_generated}/mpn/${name}.c")
  file(CONFIGURE OUTPUT "${wrapper}" CONTENT "#define OPERATION_${name} 1\n#include \"${source}\"\n" @ONLY)
  list(APPEND gmp_sources "${wrapper}")
endforeach()
add_library(gmp STATIC ${gmp_sources})

target_include_directories(gmp PUBLIC "${gmp_generated}" "${GMP_SOURCE}"
  PRIVATE "${GMP_TABLES}" "${GMP_TABLES}/mpn" "${GMP_SOURCE}/mpn/generic")

am_sources(mpfr_sources "${MPFR_SOURCE}/src" libmpfr_la_SOURCES)
# Automake's nodist source is generated by tools/get_patches.sh. Expand its
# pinned C template here so Windows builds need no shell interpreter.
file(READ "${MPFR_SOURCE}/tools/get_patches.sh" patches_script)
string(FIND "${patches_script}" "/* mpfr_get_patches" patches_start)
string(FIND "${patches_script}" "\nEOF" patches_end)
math(EXPR patches_length "${patches_end} - ${patches_start}")
string(SUBSTRING "${patches_script}" ${patches_start} ${patches_length} patches_template)
file(READ "${MPFR_SOURCE}/PATCHES" patches)
string(REGEX REPLACE "[ \t\r\n]+" " " patches "${patches}")
string(STRIP "${patches}" patches)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/get_patches.c" CONTENT "${patches_template}\n  return \"${patches}\";\n}\n" @ONLY)
list(APPEND mpfr_sources "${CMAKE_CURRENT_BINARY_DIR}/get_patches.c")
add_library(mpfr STATIC ${mpfr_sources})
target_compile_definitions(mpfr PRIVATE MPFR_TUNE_CASE="default")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mpfr-defs-${WASM_WIDTH}.json" definitions)
string(JSON count LENGTH "${definitions}")
math(EXPR last "${count} - 1")
foreach(i RANGE 0 ${last})
  string(JSON definition GET "${definitions}" ${i})
  target_compile_definitions(mpfr PRIVATE "${definition}")
endforeach()
if(WASM_THREADING)
  target_compile_definitions(mpfr PRIVATE MPFR_USE_THREAD_SAFE=1)
endif()
target_include_directories(mpfr PUBLIC "${MPFR_SOURCE}/src" PRIVATE "${MPFR_SOURCE}/src/generic")
target_link_libraries(mpfr PUBLIC gmp)
install(TARGETS gmp gmpxx mpfr ARCHIVE DESTINATION lib)

  "${MPFR_SOURCE}/src/mpfr.h" "${MPFR_SOURCE}/src/mpf2mpfr.h" DESTINATION include)
